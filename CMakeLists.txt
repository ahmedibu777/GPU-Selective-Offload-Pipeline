cmake_minimum_required(VERSION 3.20)
project(GPUSelectiveOffloadPipeline VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_executable(recorder_simulator apps/recorder_simulator/main.cpp)
target_include_directories(recorder_simulator PRIVATE ${PROJECT_SOURCE_DIR}/include)
target_compile_options(recorder_simulator PRIVATE -Wall -Wextra -Wpedantic -Werror)

include(CTest)
enable_testing()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(target recorder_simulator common_types_smoke)
    if(TARGET ${target})
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
